function renderDoctors(list = doctors) {
      resultCount.textContent = `${list.length} doctors available`;

if (!list.length) {
        doctorGrid.innerHTML = `

function selectDoctor(id) {
      currentDoctor = doctors.find(doctor => doctor.id === id);
      selectedTime = null;

        currentDoctor.initials;

        currentDoctor.name;

        currentDoctor.specialty;

    }

function selectTime(id, time) {
      const doctor = doctors.find(item => item.id === id);

if (!currentDoctor || currentDoctor.id !== id) {
        selectDoctor(id);
      }

currentDoctor = doctor;
      selectedTime = time;
      document.getElementById("selectedTime").textContent = time;
      showToast(`${time} selected`);
    }

function applyFilters() {
      const specialty = document.querySelector(
        'input[name="specialty"]:checked'
      ).value;

const consultationFilters = [
        ...document.querySelectorAll(".consultation-filter:checked")
      ].map(input => input.value);

const onlyToday = document.getElementById("availableToday").checked;
      const search = document
        .getElementById("doctorSearch")
        .value
        .toLowerCase()
        .trim();

const filtered = doctors.filter(doctor => {
        const matchesSpecialty =
          specialty === "all" || doctor.specialty === specialty;

const matchesSearch =
          !search ||
          doctor.name.toLowerCase().includes(search) ||
          doctor.specialty.toLowerCase().includes(search);

const matchesConsultation =
          !consultationFilters.length ||
          consultationFilters.some(type =>
            doctor.consultation.includes(type)
          );

const matchesAvailability =
          !onlyToday || doctor.availableToday;

return (
          matchesSpecialty &&
          matchesSearch &&
          matchesConsultation &&
          matchesAvailability
        );
      });

renderDoctors(filtered);
    }

      .querySelectorAll('input[name="specialty"], .consultation-filter')
      .forEach(input => input.addEventListener("change", applyFilters));

document
      .getElementById("availableToday")
      .addEventListener("change", applyFilters);

      .getElementById("searchForm")

        event.preventDefault();
        applyFilters();

          .getElementById("doctors")
          .scrollIntoView({ behavior: "smooth" });
      });

      .getElementById("confirmBooking")
      .addEventListener("click", () => {

          return;

          showToast("Please select an available time slot");
          return;
        }

showToast(
          `Appointment confirmed with ${currentDoctor.name} at ${selectedTime}`
        );
      });

function showToast(message) {
      toast.textContent = message;
      toast.classList.add("show");

setTimeout(() => {
        toast.classList.remove("show");
      }, 3000);
    }
